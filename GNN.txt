GATE_TYPE_MAP = {
    'NOT': 0, 'BUFF': 1, 'AND': 2, 'OR': 3,
    'NAND': 4, 'NOR': 5, 'XOR': 6, 'XNOR': 7, 'DFF': 8,
    'INPUT': 9, 'UNKNOWN': 10
}

def gates_to_gnn_input(gates, inputs, circuit_gate_count):
    """
    Builds 5-feature GNN input matching training format:
    [type_id, fan_in_norm, fan_out_norm, depth_norm, relative_size]
    """
    from core.graph_builder import build_graph
    
    # Need outputs list — use gates that nothing else consumes
    consumed = set()
    for _, (_, gate_inputs) in gates.items():
        consumed.update(gate_inputs)
    outputs = [sig for sig in gates if sig not in consumed]
    
    G        = build_graph(inputs, outputs, gates)
    nodes    = list(G.nodes())
    node_idx = {n: i for i, n in enumerate(nodes)}
    norm     = max(1.0, math.sqrt(circuit_gate_count))
    rel_size = round(circuit_gate_count / 5000.0, 4)

    # Compute depth via topological sort
    depth_map = {}
    try:
        for node in nx.topological_sort(G):
            preds = list(G.predecessors(node))
            depth_map[node] = 0 if not preds else 1 + max(
                depth_map.get(p, 0) for p in preds)
    except Exception:
        depth_map = {n: 0 for n in nodes}

    depth_norm = max(1.0, math.log2(circuit_gate_count))

    node_features = []
    edge_index    = []

    for node in nodes:
        data  = G.nodes[node]
        ntype = data.get('node_type', 'gate')
        gtype = data.get('gate_type', 'UNKNOWN')

        if ntype == 'input':
            type_id = 9   # INPUT
        else:
            type_id = GATE_TYPE_MAP.get(gtype, 10)

        fan_in  = round(G.in_degree(node)  / norm, 4)
        fan_out = round(G.out_degree(node) / norm, 4)
        depth   = round(depth_map.get(node, 0) / depth_norm, 4)

        node_features.append([type_id, fan_in, fan_out, depth, rel_size])

    for src, dst in G.edges():
        if src in node_idx and dst in node_idx:
            edge_index.append([node_idx[src], node_idx[dst]])

    return torch.tensor(node_features, dtype=torch.float32), edge_index



def gnn_simulated_annealing(circuit,
                              predictor,
                              initial_temp        = 50.0,
                              cooling_rate        = 0.95,
                              min_temp            = 0.1,
                              iterations_per_temp = 10,
                              verify_every        = 20,
                              validate            = True,
                              verbose             = True):
    """
    GNN-accelerated SA with functionally correct mutations.

    GNN replaces the cost function call in the inner loop (fast).
    Every verify_every accepted moves, real cost corrects GNN drift.
    All mutations pass validate_functional_equivalence() first.

    Returns:
        best_gates, best_cost, report_dict
    """
    current_gates  = copy.deepcopy(circuit.gates)
    current_cost   = circuit.cost
    best_gates     = copy.deepcopy(current_gates)
    best_cost      = current_cost
    temp           = initial_temp
    total_iters    = 0
    accepted       = 0
    gnn_calls      = 0
    real_calls     = 1
    verify_counter = 0

    if verbose:
        print("=" * 55)
        print("  GNN-ACCELERATED SA  (correct mutations)")
        print("=" * 55)
        print(f"  Circuit     : {circuit.name}")
        print(f"  Gates       : {circuit.gate_count}")
        print(f"  Start cost  : {round(current_cost, 4)}")
        print(f"  Validate    : {validate}")
        print(f"  Verify every: {verify_every} accepted moves")
        print("-" * 55)

    while temp > min_temp:
        for _ in range(iterations_per_temp):
            total_iters += 1

            # Correct pattern-based mutation
            new_gates = apply_safe_mutation(
                circuit.inputs, circuit.outputs, current_gates,
                max_attempts=20, validate=validate
            )
            if new_gates is None:
                continue



            node_feat, edge_index = gates_to_gnn_input(new_gates, circuit.inputs, circuit.gate_count)
            new_cost = predictor.predict(
            node_feat, edge_index, gate_count=circuit.gate_count)

            # GNN fast cost estimate
            # node_feat, edge_index = gates_to_gnn_input(
            #     new_gates, circuit.inputs
            # )

    #         node_feat, edge_index = gates_to_graph_data(
    # new_gates,
    # circuit.inputs,
    # circuit.outputs,
    # circuit.gate_count)

    #         node_feat = torch.tensor(
    #             node_feat,
    #             dtype=torch.float32
    #         )

            
            new_cost   = predictor.predict(node_feat, edge_index, gate_count=circuit.gate_count)
            gnn_calls += 1

            delta = new_cost - current_cost
            if delta < 0 or random.random() < math.exp(-delta / temp):
                current_gates   = new_gates
                current_cost    = new_cost
                accepted       += 1
                verify_counter += 1

                # Periodic correction with real cost function
                if verify_counter >= verify_every:
                    verify_counter = 0
                    verified_cost  = compute_pac_cost(
                        current_gates, circuit.inputs
                    )['total_cost']
                    real_calls   += 1
                    current_cost  = verified_cost

                if current_cost < best_cost:
                    best_cost  = current_cost
                    best_gates = copy.deepcopy(current_gates)

        temp *= cooling_rate

    # Final real-cost verification
    final_cost  = compute_pac_cost(best_gates, circuit.inputs)['total_cost']
    real_calls += 1
    improvement = (circuit.cost - final_cost) / circuit.cost * 100

    report = {
        'original_cost': round(circuit.cost, 4),
        'best_cost'    : round(final_cost, 4),
        'improvement'  : round(improvement, 2),
        'total_iters'  : total_iters,
        'accepted'     : accepted,
        'gnn_calls'    : gnn_calls,
        'real_calls'   : real_calls,
        'gnn_ratio'    : round(gnn_calls / (gnn_calls + real_calls) * 100, 1),
    }

    if verbose:
        print(f"  Iterations  : {total_iters}")
        print(f"  Accepted    : {accepted}")
        print(f"  GNN calls   : {gnn_calls}  ({report['gnn_ratio']}%)")
        print(f"  Real calls  : {real_calls}")
        print(f"  Final cost  : {round(final_cost, 4)}")
        print(f"  Improvement : {round(improvement, 2)}%")
        print("=" * 55)

    return best_gates, final_cost, report
